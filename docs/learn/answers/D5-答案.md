# D5 练习参考答案（做完再看）

```ts
export function selectPage(state: SelectionState, pageIds: string[]): SelectionState {
  if (state.selectAll) return state;
  const explicit = new Set(state.explicitIds);
  for (const id of pageIds) {
    explicit.add(id);
  }
  return { ...state, explicitIds: explicit };
}

export function clearPage(state: SelectionState, pageIds: string[]): SelectionState {
  if (state.selectAll) return state;
  const explicit = new Set(state.explicitIds);
  for (const id of pageIds) {
    explicit.delete(id);
  }
  return { ...state, explicitIds: explicit };
}
```

测试：

```ts
describe("整页操作", () => {
  const page = ["a", "b", "c"];

  it("空状态勾选整页_全部选中", () => {
    const state = selectPage(emptySelection(), page);
    expect(countSelected(state, 9999)).toBe(3);
  });

  it("已勾一条再勾整页_不重复", () => {
    const state = selectPage(toggleRow(emptySelection(), "a"), page);
    expect(countSelected(state, 9999)).toBe(3);
  });

  it("取消整页_清零", () => {
    const state = clearPage(selectPage(emptySelection(), page), page);
    expect(countSelected(state, 9999)).toBe(0);
  });

  it("全选模式_原样返回_引用相等", () => {
    const state = enterSelectAll();
    expect(selectPage(state, page)).toBe(state);
    expect(clearPage(state, page)).toBe(state);
  });
});
```

思考题答案：**"无变化"也应该体现为"无新对象"**。React（以及 TanStack Query 的结构共享）靠引用判断数据是否变了——内容相等但引用不同会触发一次没有意义的重渲染。全选模式下整页操作是 no-op，返回原引用让 React 跳过更新，这是把"语义上的无变化"翻译成"渲染上的无变化"。

顺带：`new Set(existing)` + 增删再包新对象，是不可变更新的标准姿势；直接 `state.explicitIds.add(id)` 会 mutate 旧状态——所有依赖旧引用的地方（比如已渲染的行）都会被波及，这在 React 里是最常见的"玄学 bug"来源。
